cmake_minimum_required(VERSION 3.24)
project(unspool_backend VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)

FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.28.0 GIT_SHALLOW TRUE)
set(HTTPLIB_REQUIRE_OPENSSL OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_TEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(httplib)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)

add_library(unspool_core src/static_security.cpp)
target_include_directories(unspool_core PUBLIC include)
target_link_libraries(unspool_core PUBLIC OpenSSL::Crypto)
if(WIN32)
  target_compile_definitions(unspool_core PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00)
endif()
if(MSVC)
  target_compile_options(unspool_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(unspool_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(unspool_server src/main.cpp)
target_link_libraries(unspool_server PRIVATE unspool_core httplib::httplib)

enable_testing()
add_executable(unspool_backend_tests tests/static_security_test.cpp)
target_link_libraries(unspool_backend_tests PRIVATE unspool_core)
add_test(NAME unspool_backend_tests COMMAND unspool_backend_tests)
